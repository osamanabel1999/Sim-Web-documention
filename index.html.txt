        function applyEfbBaseStyleEnvironment() {
            try {
                if (state.mapStyle === 'SATELLITE') {
                    // Satellite-v9 is an imagery base. Do not apply the dark
                    // atmospheric fog used by the dark aviation style.
                    map.setFog(null);
                } else {
                    map.setFog({
                        'color': 'rgb(4, 10, 24)',
                        'high-color': 'rgb(36, 92, 223)',
                        'horizon-blend': 0.1,
                        'space-color': 'rgb(2, 4, 10)',
                        'star-intensity': 0.7
                    });
                }
            } catch (e) {}
        }

        function applyEfbRuntimeLighting() {
            if (!map || !map.isStyleLoaded()) return;

            // Mapbox GL JS 3.x can apply style lighting to custom layers.
            // Emissive strength 1 makes Flutter-owned overlays retain their
            // authored colors instead of becoming darkened by the basemap.
            EFB_RUNTIME_LINE_LAYERS.forEach(function(id) {
                try {
                    if (map.getLayer(id)) {
                        map.setPaintProperty(id, 'line-emissive-strength', 1);
                    }
                } catch (e) {}
            });

            EFB_RUNTIME_FILL_LAYERS.forEach(function(id) {
                try {
                    if (map.getLayer(id)) {
                        map.setPaintProperty(id, 'fill-emissive-strength', 1);
                    }
                } catch (e) {}
            });

            EFB_RUNTIME_CIRCLE_LAYERS.forEach(function(id) {
                try {
                    if (map.getLayer(id)) {
                        map.setPaintProperty(id, 'circle-emissive-strength', 1);
                    }
                } catch (e) {}
            });

            EFB_RUNTIME_SYMBOL_LAYERS.forEach(function(id) {
                try {
                    if (map.getLayer(id)) {
                        map.setPaintProperty(id, 'text-emissive-strength', 1);
                    }
                } catch (e) {}
                try {
                    if (map.getLayer(id)) {
                        map.setPaintProperty(id, 'icon-emissive-strength', 1);
                    }
                } catch (e) {}
            });

            // The Flutter bridge already owns the exact operational colors.
            // Re-apply them after the new base style and after Flutter has
            // rebuilt its runtime layers.
            try {
                if (typeof window.forceEfbOperationalLayerColors === 'function') {
                    window.forceEfbOperationalLayerColors();
                }
            } catch (e) {}
        }

        function stabilizeEfbRuntimeOverlays() {
            // Style replacement + WebView/Flutter injection can finish on
            // different ticks. Apply on several short ticks so no base-style
            // operation can leave the Flutter overlays in a stale visual state.
            const delays = [0, 60, 180, 400, 800];
            delays.forEach(function(delay) {
                setTimeout(function() {
                    try {
                        applyEfbBaseStyleEnvironment();
                        applyEfbRuntimeLighting();
                    } catch (e) {}
                }, delay);
            });
        }

        async function fetchAirports() {
            try {
                const res = await fetch('https://gist.githubusercontent.com/tdreyno/4278655/raw/airports.json');
                rawAirports = await res.json();
                updateAirportsStyle();
            } catch (e) {}
        }

        async function fetchPlanes() {
            let features = [];
            
            if (state.vatsim) {
                try {
                    const res = await fetch('https://data.vatsim.net/v3/vatsim-data.json');
                    const data = await res.json();
                    data.pilots.forEach(p => {
                        features.push(createPlaneFeature(p, 'VATSIM', true, '#FFC107', p.cid.toString()));
                    });
                } catch (e) {}
            }
            
            if (state.ivao) {
                try {
                    const res = await fetch('https://api.ivao.aero/v2/tracker/whazzup');
                    const data = await res.json();
                    data.clients.pilots.forEach(p => {
                        features.push(createPlaneFeature(p, 'IVAO', false, '#69F0AE', p.userId.toString()));
                    });
                } catch (e) {}
            }

            planesGeoJSON.features = features;
            if (map.getSource('planes')) {
                map.getSource('planes').setData(planesGeoJSON);
            }
        }

        function createPlaneFeature(p, net, isVatsim, baseColor, id) {
            const lat = isVatsim ? p.latitude : p.lastTrack?.latitude;
            const lon = isVatsim ? p.longitude : p.lastTrack?.longitude;
            const hdg = isVatsim ? p.heading : p.lastTrack?.heading;
            const alt = isVatsim ? p.altitude : p.lastTrack?.altitude;
            const gs = isVatsim ? p.groundspeed : p.lastTrack?.groundSpeed;
            const vs = isVatsim ? p.vertical_speed : p.lastTrack?.verticalSpeed;
            const color = (state.userId && state.userId === id) ? '#FF5252' : baseColor;

            return {
                type: 'Feature',
                properties: {
                    id: id,
                    callsign: p.callsign || 'N/A',
                    net: net,
                    isVatsim: isVatsim,
                    hdg: Math.round(hdg || 0),
                    alt: Math.round(alt || 0),
                    gs: Math.round(gs || 0),
                    vs: Math.round(vs || 0),
                    color: color,
                    isUser: (state.userId && state.userId === id),
                    raw: p // تخزين الـ JSON الكامل لبعثه لـ Flutter
                },
                geometry: { type: 'Point', coordinates: [lon || 0, lat || 0] }
            };
        }

        function updateAirportsStyle() {
            if (rawAirports.length === 0) return;
            let features = [];
            
            rawAirports.forEach(ap => {
                const icao = ap.icao || "";
                const isTarget = (icao === state.depIcao || icao === state.arrIcao);
                // عرض المطار لو زرار المطارات شغال، أو لو المطار ده هو نقطة بداية/نهاية المسار
                if (state.airports || (state.flightPath && isTarget)) {
                    features.push({
                        type: 'Feature',
                        properties: { icao: icao, isTarget: isTarget, raw: ap },
                        geometry: { type: 'Point', coordinates: [parseFloat(ap.lon), parseFloat(ap.lat)] }
                    });
                }
            });

            if (map.getSource('airports')) {
                map.getSource('airports').setData({ type: 'FeatureCollection', features: features });
            }
        }

        function drawFlightPath() {
            if (!state.flightPath || !state.depIcao || !state.arrIcao || rawAirports.length === 0) {
                if (map.getSource('flight-path')) map.getSource('flight-path').setData({ type: 'FeatureCollection', features: [] });
                return;
            }

            const dep = rawAirports.find(a => a.icao === state.depIcao);
            const arr = rawAirports.find(a => a.icao === state.arrIcao);

            if (dep && arr) {
                // رسم الخط المقوس بواسطة Turf.js
                const start = [parseFloat(dep.lon), parseFloat(dep.lat)];
                const end = [parseFloat(arr.lon), parseFloat(arr.lat)];
                const greatCircle = turf.greatCircle(start, end);
                if (map.getSource('flight-path')) map.getSource('flight-path').setData(greatCircle);
            }
        }

        function updateVisibility() {
            if(!map.getLayer('airports-circle')) return;
            
            map.setLayoutProperty('airports-circle', 'visibility', (state.airports || state.flightPath) ? 'visible' : 'none');
            map.setLayoutProperty('airports-label', 'visibility', (state.airports || state.flightPath) ? 'visible' : 'none');
            
            // Flutter owns the operational Flight Path. Keep the legacy
            // HTML compatibility layer hidden so it can never compete with
            // the Flutter-colored layer after a style switch.
            map.setLayoutProperty('flight-path-layer', 'visibility', 'none');
            
            map.setLayoutProperty('planes-callsign', 'visibility', (state.callsigns && !state.radar) ? 'visible' : 'none');
            map.setLayoutProperty('planes-radar', 'visibility', state.radar ? 'visible' : 'none');
        }

        function updateWeather() {
            if(!map.getLayer('layer-rain')) return;
            map.setLayoutProperty('layer-rain', 'visibility', state.weather === 'RAIN' ? 'visible' : 'none');
            map.setLayoutProperty('layer-temp', 'visibility', state.weather === 'TEMP' ? 'visible' : 'none');
            map.setLayoutProperty('layer-wind', 'visibility', state.weather === 'WIND' ? 'visible' : 'none');
        }

        // ==========================================
        // 5. التفاعل بالضغط (Clicks to Flutter)
        // ==========================================

        map.on('click', 'planes-icon', (e) => {
            if (e.features.length > 0) {
                const prop = e.features[0].properties;
                sendToFlutter('PLANE_CLICKED', { type: 'PLANE', net: prop.net, isVatsim: prop.isVatsim, data: JSON.parse(prop.raw) });
            }
        });

        map.on('click', 'airports-circle', (e) => {
            if (e.features.length > 0) {
                const prop = e.features[0].properties;
                sendToFlutter('AIRPORT_CLICKED', { type: 'AIRPORT', data: JSON.parse(prop.raw) });
            }
        });

        map.on('click', (e) => {
            // تجاهل لو الضغطة كانت على طيارة أو مطار
            const features = map.queryRenderedFeatures(e.point, { layers: ['planes-icon', 'airports-circle'] });
            if (features.length > 0) return;

            if (state.teleportMode) {
                sendToFlutter('TELEPORT_CLICKED', { lat: e.lngLat.lat, lon: e.lngLat.lng });
            } else {
                sendToFlutter('MAP_CLICKED', {});
            }
        });

        // تغيير شكل الماوس عند المرور فوق العناصر
        map.on('mouseenter', 'planes-icon', () => map.getCanvas().style.cursor = 'pointer');
        map.on('mouseleave', 'planes-icon', () => map.getCanvas().style.cursor = '');
        map.on('mouseenter', 'airports-circle', () => map.getCanvas().style.cursor = 'pointer');
        map.on('mouseleave', 'airports-circle', () => map.getCanvas().style.cursor = '');

        // تحديث حي للطائرات كل 30 ثانية
        setInterval(fetchPlanes, 30000);

    </script>
